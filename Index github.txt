    async function calculateNewRoster() {
      showModal("กำลังเชื่อมโยงข้อมูล...", "ดึงข้อมูลล่าสุดจาก Google Sheet และ Co-Work Sync");
      var month = parseInt(document.getElementById("monthSelect").value);

      try {
        let response = await fetch(`${GAS_API_URL}?action=getRoster&year=2026&month=${month}`);
        let data = await response.json();
        currentData = data;
        renderWebPage();
        updateSyncStatusUI(data.isFromSheet, data.prevContext);
        showModalSuccess("สำเร็จ!", data.isFromSheet ? "ดึงข้อมูลจาก Google Sheet สำเร็จ" : "สร้างตารางใหม่เรียบร้อย");
      } catch (error) {
        statusModal.hide();
        alert("เกิดข้อผิดพลาดในการโหลดข้อมูล: " + error);
      }
    }

    async function silentBackgroundSync() {
      var month = parseInt(document.getElementById("monthSelect").value);
      try {
        let response = await fetch(`${GAS_API_URL}?action=getRoster&year=2026&month=${month}`);
        let data = await response.json();
        if (data && currentData) {
          var isDifferent = JSON.stringify(data.roster) !== JSON.stringify(currentData.roster);
          if (isDifferent) {
            currentData = data;
            renderWebPage();
            showToast("🔄 ตรวจพบการเปลี่ยนแปลง! ซิงก์ข้อมูลให้เป็นปัจจุบันแล้ว");
          }
        }
      } catch (e) {
        console.error("Background sync failed", e);
      }
    }

    async function generateNewRosterConfirm() {
      if (confirm("คุณต้องการล้างและจัดตารางใหม่ด้วยระบบอัตโนมัติใช่หรือไม่?")) {
        showModal("กำลังจัดตารางใหม่...", "ระบบกำลังคำนวณโควตาใหม่ทั้งหมด");
        var month = parseInt(document.getElementById("monthSelect").value);
        
        try {
          let response = await fetch(`${GAS_API_URL}?action=generateNewRoster&year=2026&month=${month}`);
          let data = await response.json();

        } catch (error) {
          statusModal.hide();
          alert("เกิดข้อผิดพลาด: " + error);
        }
      }
    }

    async function applyCustomQuotasAndRegenerate() {
      if (!currentData) return;
      var customQuotas = {};

      currentData.summary.forEach(function(s) {
        if (editableMainOfficers.includes(s.name)) {
          var inputMainEl = document.getElementById("tune_main_" + s.name);
          if (inputMainEl) {
            customQuotas["main_" + s.name] = parseInt(inputMainEl.value) || s.main;
          }
        }
        if (assistantOfficers.includes(s.name)) {
          var inputSleepEl = document.getElementById("tune_sleep_" + s.name);
          if (inputSleepEl) {
            customQuotas["sleep_" + s.name] = parseInt(inputSleepEl.value) || s.sleep;
          }
        }
      });

      showModal("กำลังประมวลผลตาราง...", "ระบบกำลังปรับเกลี่ยตามโควตาที่คุณกำหนด");
      var month = parseInt(document.getElementById("monthSelect").value);

      try {
        let response = await fetch(`${GAS_API_URL}`, {
          method: 'POST',
          body: JSON.stringify({
            action: 'generateWithQuotas',
            year: 2026,
            month: month,
            customQuotas: customQuotas
          })
        });
        let data = await response.json();
        currentData = data;
        renderWebPage();
        showModalSuccess("อัปเดตสำเร็จ!", "ปรับตารางเวรตามโควตาเรียบร้อยแล้ว");
        showToast("อัปเดตตารางเวรสำเร็จ!");
      } catch (error) {
        statusModal.hide();
        alert("เกิดข้อผิดพลาด: " + error);
      }
    }

      }
    }

    function renderWebPage() {
      if (!currentData) return;

      var dHtml = "";
      currentData.roster.forEach(function(row, idx) {
        dHtml += "<tr>" +
          "<td><b>" + row.day + "</b></td>" +
          "<td>" + row.dayName + "</td>" +
          "<td>" + createDropdown(idx, 'main', row.main, listMainOfficers) + "</td>" +
          "<td>" + createDropdown(idx, 'asst1', row.asst1, listAsst1) + "</td>" +
          "<td>" + createDropdown(idx, 'asst2', row.asst2, listAsst2) + "</td>" +
          "<td>" + createDropdown(idx, 'sleep', row.sleep, listSleep) + "</td>" +
          "</tr>";
      });
      document.getElementById("dutyBody").innerHTML = dHtml;

      renderSummaryTable();
      renderWeekendStatsTable();
    }

    function createDropdown(rowIndex, colType, selectedName, relevantList) {
      var style = colorMap[selectedName] || { bg: "#ffffff", text: "#000000" };
      var html = '<select class="form-select select-name" style="background-color:' + style.bg + '; color:' + style.text + ';" onchange="updateDutyPerson(' + rowIndex + ', \'' + colType + '\', this.value)">';
      relevantList.forEach(function(name) {
        var selected = (name === selectedName) ? "selected" : "";
        html += '<option value="' + name + '" ' + selected + '>' + name + '</option>';
      });
      html += '</select>';
      return html;
    }

    function updateDutyPerson(rowIndex, colType, newName) {
      currentData.roster[rowIndex][colType] = newName;
      recalculateSummaryInMemory();
      renderSummaryTable();
      renderWeekendStatsTable();
      showToast("แก้ไขรายชื่อเรียบร้อย!");
    }

    function recalculateSummaryInMemory() {
      var summaryMap = {};
      var allNames = currentData.allOfficers;
      allNames.forEach(function(name) {
        summaryMap[name] = { main: 0, asst: 0, sleep: 0, total: 0 };
      });

      currentData.roster.forEach(function(row) {
        if (summaryMap[row.main]) summaryMap[row.main].main++;
        if (summaryMap[row.asst1]) summaryMap[row.asst1].asst++;
        if (summaryMap[row.asst2]) summaryMap[row.asst2].asst++;
        if (summaryMap[row.sleep]) summaryMap[row.sleep].sleep++;
      });

      currentData.summary = [];
      allNames.forEach(function(name) {
        var item = summaryMap[name];
        currentData.summary.push({
          name: name,
          main: item.main,
          asst: item.asst,
          sleep: item.sleep,
          total: item.main + item.asst + item.sleep
        });
      });

      var weekendStatsMap = {};
      listMainOfficers.forEach(function(name) {
        weekendStatsMap[name] = { mainCount: 0, mainDates: [], asst2Count: 0, asst2Dates: [] };
      });

      currentData.roster.forEach(function(row) {
        if (row.dayName === "วันเสาร์" || row.dayName === "วันอาทิตย์") {
          var m = row.main;
          if (weekendStatsMap[m]) {
            weekendStatsMap[m].mainCount++;
            weekendStatsMap[m].mainDates.push(row.day + "(" + row.dayName.replace("วัน", "") + ")");
          }
          var a2 = row.asst2;
          if (weekendStatsMap[a2]) {
            weekendStatsMap[a2].asst2Count++;
            weekendStatsMap[a2].asst2Dates.push(row.day + "(" + row.dayName.replace("วัน", "") + ")");
          }
        }
      });

      currentData.weekendStats = {};
      listMainOfficers.forEach(function(name) {
        currentData.weekendStats[name] = {
          mainWeekendCount: weekendStatsMap[name].mainCount,
          mainDates: weekendStatsMap[name].mainDates,
          asst2WeekendCount: weekendStatsMap[name].asst2Count,
          asst2Dates: weekendStatsMap[name].asst2Dates
        };
      });
    }

        sHtml += "<tr>" +

            sHtml += "<td class='text-center'>" +

              "id='tune_main_" + s.name + "' value='" + s.main + "' min='0' max='31'>" +
              "</td>";

            sHtml += "<td class='text-center'>" + s.main + " วัน <span class='text-muted small'>(เฉลี่ย)</span></td>";
          } else {
            sHtml += "<td class='text-center'>" + s.main + " วัน</td>";
          }

          sHtml += "<td class='text-center'>" + s.asst + " วัน</td>";

          if (assistantOfficers.includes(s.name)) {

              "<input type='number' class='form-control form-control-sm input-summary-tune d-inline-block' " +

            sHtml += "<td class='text-center text-muted'>-</td>";

    }

    async function saveToGoogleSheet() {
      if (!currentData) return;
      showModal("กำลังบันทึกและซิงก์...", "บันทึกข้อมูลตารางเวรทั้งหมดลง Google Sheet");

      try {
        let response = await fetch(`${GAS_API_URL}`, {
          method: 'POST',
          body: JSON.stringify({
            action: 'saveRoster',
            finalData: currentData
          })
        });
        let res = await response.json();
        showModalSuccess("บันทึกสำเร็จ!", res.message);
        document.getElementById("lblSyncStatus").innerHTML = "<b class='text-success'><i class='fa-solid fa-cloud-arrow-down me-1'></i>เชื่อมโยงสมบูรณ์กับ Google Sheet (Vercel Co-Work Active)</b>";
      } catch (error) {
        statusModal.hide();
        alert("เกิดข้อผิดพลาดในการบันทึก: " + error);
      }
    }
